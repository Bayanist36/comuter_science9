function buildChartLeaks(){

  const data=[{v:60,l:'пароли',c:'#FF6EC7'},{v:40,l:'прочее',c:'#3D4E86'}];
  const W=430,H=330,R=142,RI=92;
  const svg=d3.select('#chart-leaks').append('svg').attr('viewBox',`0 0 ${W} ${H}`);
  const g=svg.append('g').attr('transform',`translate(${W/2},${H/2})`);
  const arc=d3.arc().innerRadius(RI).outerRadius(R);
  const pie=d3.pie().sort(null).padAngle(.025).value(d=>d.v);
  const dur=REDUCED?0:1100;
  const paths=g.selectAll('path').data(pie(data)).join('path')
    .attr('fill',d=>d.data.c).attr('stroke','rgba(7,11,24,.9)').attr('stroke-width',2)
    .attr('d',arc);
  if(!REDUCED){
    paths.transition().duration(dur).ease(d3.easeCubicOut)
      .attrTween('d',function(d){const i=d3.interpolate({startAngle:d.startAngle,endAngle:d.startAngle},d);return t=>arc(i(t));});
  }
  g.append('text').attr('text-anchor','middle').attr('y',-4)
    .attr('fill','#EEF3FF').style('font','800 54px Unbounded').text('60 %');
  g.append('text').attr('text-anchor','middle').attr('y',30)
    .attr('fill','#A9B6DC').style('font','600 17px Manrope').text('утечек — из-за паролей');
  const lab=g.selectAll('text.sl').data(pie(data)).join('text').attr('class','sl')
    .attr('transform',d=>`translate(${arc.centroid(d)})`)
    .attr('text-anchor','middle').attr('fill','#071120')
    .style('font','800 19px Manrope').attr('opacity',REDUCED?1:0)
    .text((d,i)=>i===0?'60 %':'40 %');
  lab.transition().delay(dur*.7).duration(REDUCED?0:400).attr('opacity',1);
}

(function(){
  const sleep=ms=>new Promise(r=>setTimeout(r,ms));
  async function run(){
    await sleep(1800);
    if(typeof Reveal==='undefined'){ document.getElementById('audit-out').textContent='AUDIT:NO_REVEAL'; return; }
    const n=Reveal.getTotalSlides();
    const res=[];
    for(let i=0;i<n;i++){
      Reveal.slide(i); await sleep(700);
      const s=Reveal.getCurrentSlide();
      const sb=s.getBoundingClientRect();
      const r={id:s.id, sh:s.scrollHeight, ch:s.clientHeight, over:[]};
      s.querySelectorAll('*').forEach(el=>{
        if(el.closest('aside.notes')) return;
        const b=el.getBoundingClientRect();
        if(b.height<2||b.width<2) return;
        const cls=(typeof el.className==='string'&&el.className)?el.className:el.tagName;
        if(b.bottom>sb.bottom+2) r.over.push(cls+' B+'+Math.round(b.bottom-sb.bottom));
        if(b.right>sb.right+2) r.over.push(cls+' R+'+Math.round(b.right-sb.right));
        if(el.scrollHeight>el.clientHeight+2 && getComputedStyle(el).overflow!=='visible') r.over.push(cls+' INNER '+el.scrollHeight+'/'+el.clientHeight);
      });
      res.push(r);
    }
    document.getElementById('audit-out').textContent='AUDIT:'+JSON.stringify(res);
  }
  window.addEventListener('load',run);
})();
</script>
